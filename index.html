public class ShopeePriceMonitor {
    public void fetchPrice(String url) {
        Document doc = Jsoup.connect(url)
            .userAgent("Mozilla/5.0")
            .timeout(5000).get();

        String priceStr = doc.select(".pqTWkA").text();
        double price = Double.parseDouble(priceStr);
        
        if (price < targetThreshold) {
            sendTelegramAlert("Price drop: RM" + price);
        }
    }
}

ASP.NET C# 
[HttpPost("api/orders/create")]
public async Task<IActionResult> CreateOrder([FromBody] OrderReq req) {
    var order = new Order {
        TableNumber = req.TableNum,
        Status = OrderStatus.Preparing,
        CreatedAt = DateTime.UtcNow
    };
    await _db.Orders.AddAsync(order);
    await _db.SaveChangesAsync();
    return Ok(new { OrderId = order.Id });
}
